Query SQL — Gate Out no dia atual

SELECT r.rota_id, CAST(r.data_operacao AS DATE) AS data,
    r.facility, r.cycle, r.wave, r.loyalty,
    t.nome_transportadora AS transportadora, r.modal,
    CASE WHEN g.yms_gate_out IS NULL THEN 'Sem Bip'
         WHEN g.yms_gate_out > r.oc_planned_end_time THEN 'Delay'
         ELSE 'Aderente' END AS status,
    DATEDIFF(MINUTE, r.oc_planned_end_time, g.yms_gate_out) AS atraso_min,
    r.oc_planned_end_time AS fim_planejado,
    g.yms_gate_out AS gate_out, g.causa_processo, g.causa_tempo
FROM rotas r
LEFT JOIN transportadoras t ON t.id = r.transportadora_id
LEFT JOIN yms_gate_out g ON g.rota_id = r.rota_id
WHERE CAST(r.data_operacao AS DATE) = CAST(GETDATE() AS DATE)
  AND r.facility = 'SSC8'
ORDER BY r.cycle, g.yms_gate_out NULLS LAST;
-- Oracle: GETDATE()→TRUNC(SYSDATE) | DATEDIFF→(ts2-ts1)*1440
